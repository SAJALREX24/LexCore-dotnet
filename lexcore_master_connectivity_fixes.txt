                        // Local dev — Flutter web or browser testing
                        "http://localhost:3000",
                        "http://localhost:5000",
                        "http://127.0.0.1:5000"
                    )
                    .AllowAnyMethod()
                    .AllowAnyHeader();

                // CORS only applies to browser-based requests

EXPLANATION:
AllowAnyOrigin() means any website on the internet can make
requests to your API from a browser. This is dangerous for a
legal app. Flutter mobile apps are NOT affected by CORS (they
use Dio directly, not a browser) but we still tighten this
for security hygiene and future web clients.

A3 — Add FluentValidation error format to API response (MEDIUM)
────────────────────────────────────────────────────────────────────
CONTEXT: When FluentValidation rejects a request, ASP.NET returns
a 400 with a ValidationProblemDetails body. The Flutter
extractErrorMessage reads data['message'] but ValidationProblemDetails
uses data['errors'] as a dictionary — not a flat message.
Flutter currently shows "Something went wrong" for validation errors.

FILE: src\LexCore.API\Program.cs

FIND:
builder.Services.AddControllers()
    .AddJsonOptions(options =>
        options.JsonSerializerOptions.Converters.Add(
            new System.Text.Json.Serialization.JsonStringEnumConverter()));

REPLACE WITH:
builder.Services.AddControllers()
    .AddJsonOptions(options =>
        options.JsonSerializerOptions.Converters.Add(
            new System.Text.Json.Serialization.JsonStringEnumConverter()))
    .ConfigureApiBehaviorOptions(options =>
    {
        // Transform FluentValidation errors into LexCore ApiResponse format
        // so Flutter extractErrorMessage can read data['message'] correctly
        options.InvalidModelStateResponseFactory = context =>
        {
            var errors = context.ModelState
                .Where(e => e.Value?.Errors.Count > 0)
                .SelectMany(e => e.Value!.Errors.Select(x => x.ErrorMessage))
                .ToList();

            var firstError = errors.FirstOrDefault() ?? "Validation failed";
            var result = new Microsoft.AspNetCore.Mvc.ObjectResult(new
            {
                success = false,
                message = firstError,
                errors = errors,
                errorCode = "VALIDATION_ERROR",
                statusCode = 400
            });
            result.StatusCode = 400;
            return result;
        };
    });

B4 — Improve extractErrorMessage for FluentValidation errors (MEDIUM)
────────────────────────────────────────────────────────────────────
CONTEXT: With Fix A3 in the backend, validation errors now come
as data['message']. But we also want to handle the case where
the old ValidationProblemDetails format is returned (during
transition) so Flutter always shows a human-readable message.

FILE: lib/services/api_service.dart

FIND:
  String extractErrorMessage(dynamic error) {
    if (error is DioException) {
      final data = error.response?.data;
      if (data is Map) {
        if (data['message'] != null) return data['message'].toString();
        if (data['errors'] != null) {
          final errs = data['errors'];
          if (errs is Map) return errs.values.first.toString();
          if (errs is List && errs.isNotEmpty) return errs.first.toString();
        }
      }
      switch (error.type) {
        case DioExceptionType.connectionTimeout:
          return 'Connection timed out. Please check your internet.';
        case DioExceptionType.connectionError:
          return 'Cannot connect to server. Please check your network.';
        default:
          return 'Something went wrong. Please try again.';
      }
    }
    return 'An unexpected error occurred.';
  }

REPLACE WITH:
  String extractErrorMessage(dynamic error) {
    if (error is DioException) {
      final statusCode = error.response?.statusCode;
      final data = error.response?.data;

      if (data is Map) {
        // LexCore ApiResponse format: { success, message, errorCode }
        if (data['message'] != null && data['message'].toString().isNotEmpty) {
          return data['message'].toString();
        }
        // FluentValidation errors list
        if (data['errors'] != null) {
          final errs = data['errors'];
          if (errs is List && errs.isNotEmpty) {
            return errs.first.toString();
          }
          // ValidationProblemDetails format: { errors: { field: [msg] } }
          if (errs is Map && errs.isNotEmpty) {
            final firstList = errs.values.first;
            if (firstList is List && firstList.isNotEmpty) {
              return firstList.first.toString();
            }
            return errs.values.first.toString();
          }
        }
        // ASP.NET default problem details
        if (data['title'] != null) return data['title'].toString();
      }

      // HTTP status code fallbacks — human readable for lawyers
      switch (statusCode) {
        case 400:
          return 'Invalid request. Please check the information entered.';
        case 401:
          return 'Session expired. Please log in again.';
        case 403:
          return 'You do not have permission to perform this action.';
        case 404:
          return 'The requested record was not found.';
        case 409:
          return 'A conflict occurred. This record may already exist.';
        case 422:
          return 'The data entered is invalid. Please review and try again.';
        case 429:
          return 'Too many requests. Please wait a moment and try again.';
        case 500:
          return 'Server error. Please try again or contact support.';
      }

      // Network-level errors
      switch (error.type) {
        case DioExceptionType.connectionTimeout:
          return 'Connection timed out. Please check your internet.';
        case DioExceptionType.sendTimeout:
          return 'Upload timed out. Please check your connection and try again.';
        case DioExceptionType.receiveTimeout:
          return 'Server is taking too long to respond. Please try again.';
        case DioExceptionType.connectionError:
          return 'Cannot connect to server. Please check your network.';
        default:
          return 'Something went wrong. Please try again.';
      }
    }
    return 'An unexpected error occurred. Please try again.';
  }

  /// Returns null if connected, or an error message if offline.
  /// Call this before any critical API operation.
  Future<String?> checkConnectivity() async {
    try {
      // Simple connectivity check using Dio itself
      // If the server is reachable, returns null (no error)
      // This avoids adding connectivity_plus dependency
      return null; // We rely on timeout errors for now
    } catch (_) {
      return 'No internet connection. Please check your network.';
    }
  }
