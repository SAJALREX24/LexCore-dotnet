  String _feeTypeDescription(String feeType) {
    switch (feeType) {
      case 'Fixed Fee':
        return 'A single agreed amount for the entire case regardless of hearings.';
      case 'Per Hearing':
        return 'Fees are charged per court appearance. Set the amount per hearing below.';
      case 'Retainer':
        return 'A recurring monthly fee to retain your services for this client.';
      case 'Contingency':
        return 'Fees are payable only upon successful outcome. No upfront charge.';
